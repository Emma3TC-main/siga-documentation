Project SIGA_DER_02 {
  database_type: "PostgreSQL"
  Note: '''
  DER-02 — Modelo lógico global por bounded contexts.

  FK sólidas: solo dentro del mismo schema/ownership.
  Relaciones entre bounded contexts: UUID externo, evento o proyección eventual.
  PostgreSQL no contiene FK cross-schema entre microservicios.
  '''
}

/* ============================================================
   IAM — identity-service
   ============================================================ */
Table iam.user_account {
  id uuid [pk]
  username varchar(100)
  email varchar(254)
  password_hash varchar(255)
  active boolean
  mfa_enabled boolean
  mfa_secret_encrypted text
  failed_login_attempts integer
  locked_until timestamptz
  version bigint
  created_at timestamptz
  updated_at timestamptz

  Note: '''
  Identity principal.
  No FK cross-schema hacia otros bounded contexts.
  '''
}

Table iam.role {
  id uuid [pk]
  code varchar(80)
  name varchar(120)
  active boolean
  version bigint
  created_at timestamptz
  updated_at timestamptz
}

Table iam.permission {
  id uuid [pk]
  code varchar(120)
  description varchar(255)
  created_at timestamptz
}

Table iam.user_role {
  user_id uuid
  role_id uuid
  assigned_at timestamptz

  indexes {
    (user_id, role_id) [pk]
  }
}

Table iam.role_permission {
  role_id uuid
  permission_id uuid
  assigned_at timestamptz

  indexes {
    (role_id, permission_id) [pk]
  }
}

Table iam.refresh_token {
  id uuid [pk]
  user_id uuid
  token_hash varchar(255)
  jti uuid
  family_id uuid
  expires_at timestamptz
  revoked_at timestamptz
  replaced_by_jti uuid
  created_at timestamptz
}

Table iam.outbox_event {
  id uuid [pk]
  aggregate_type varchar(80)
  aggregate_id uuid

  schema_version integer
